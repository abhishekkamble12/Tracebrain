- [ ] 3. **P2-A: Create MemoryContext and wire it into AppShell**

      Create a new context following the exact pattern of `SelectionContext.tsx`. This eliminates the three independent localStorage reads/writes scattered across the codebase.

      **Create `d:\Gravel\ui\src\state\MemoryContext.tsx`:**
      ```tsx
      import React, { createContext, useContext, useState, useEffect } from 'react';

      export interface MemoryState {
        memoryEnabled: boolean;
        setMemoryEnabled: (v: boolean) => void;
      }

      const MemoryContext = createContext<MemoryState | null>(null);

      export const MemoryProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
        const [memoryEnabled, setMemoryEnabledState] = useState<boolean>(() => {
          try {
            return localStorage.getItem('tracebrain.memoryEnabled') !== 'false';
          } catch {
            return true;
          }
        });

        useEffect(() => {
          try {
            localStorage.setItem('tracebrain.memoryEnabled', String(memoryEnabled));
          } catch { /* ignore */ }
        }, [memoryEnabled]);

        const setMemoryEnabled = (v: boolean) => setMemoryEnabledState(v);

        return (
          <MemoryContext.Provider value={{ memoryEnabled, setMemoryEnabled }}>
            {children}
          </MemoryContext.Provider>
        );
      };

      export function useMemory(): MemoryState {
        const ctx = useContext(MemoryContext);
        if (!ctx) throw new Error('useMemory must be used within MemoryProvider');
        return ctx;
      }
      ```

      **AppShell.tsx** — import `MemoryProvider` and wrap:

      ```tsx
      // Wrap the return value:
      return (
        <MemoryProvider>
          <div className="min-h-screen ...">
            ...
          </div>
        </MemoryProvider>
      );
      ```

      TypeScript: `MemoryState` interface exported for consumers. No changes to existing types needed.

- [ ] 17. **P1-A: Create TimeScrubber component**

       **Create `d:\Gravel\ui\src\features\graph\TimeScrubber.tsx`:**

       ```tsx
       import React from 'react';
       import { useSelection } from '@/state/SelectionContext';
       import { cn } from '@/lib/utils';

       // 56-day window, epoch reference: arbitrary base date used for display
       const DAY_MS = 86_400_000;
       const WINDOW_DAYS = 56;

       // Known topology-mutation events (day offsets from window start)
       const EVENT_PINS = [
         { day: 24, label: 'currency → currency-v2' },
         { day: 36, label: 'payments ownership → Team Alpha' },
         { day: 44, label: 'recommendation → ad cutover' },
       ] as const;

       // Base date: 56 days before "now" — kept stable per render via module-level const
       const BASE_DATE = new Date(Date.now() - WINDOW_DAYS * DAY_MS);

       function dayToTimestamp(day: number): number {
         return Math.floor((BASE_DATE.getTime() + day * DAY_MS) / 1000); // seconds
       }

       function formatDayDate(day: number): string {
         const d = new Date(BASE_DATE.getTime() + day * DAY_MS);
         return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
       }

       export const TimeScrubber: React.FC = () => {
         const { scrubbedTimestamp, setScrubbedTimestamp, activeTopoMode, setTopoMode } = useSelection();

         // Derive current day from scrubbedTimestamp
         const currentDay = React.useMemo(() => {
           if (!scrubbedTimestamp) return WINDOW_DAYS; // default: rightmost (now)
           const ts = scrubbedTimestamp * 1000; // seconds → ms
           const day = Math.round((ts - BASE_DATE.getTime()) / DAY_MS);
           return Math.max(0, Math.min(WINDOW_DAYS, day));
         }, [scrubbedTimestamp]);

         // Debounced update ref
         const debounceRef = React.useRef<ReturnType<typeof setTimeout> | null>(null);

         const handleChange = (e: React.ChangeEvent<HTMLInputElement>) => {
           const day = Number(e.target.value);
           if (debounceRef.current) clearTimeout(debounceRef.current);
           debounceRef.current = setTimeout(() => {
             setScrubbedTimestamp(dayToTimestamp(day));
           }, 250);
         };

         const handleJumpNow = () => {
           setScrubbedTimestamp(null);
           setTopoMode('now');
         };

         const handleAsOf = () => {
           setTopoMode('asof');
         };

         return (
           <div className="flex items-center gap-2 px-4 py-1.5 bg-[var(--surface-panel)]/70 border-t border-[var(--border-subtle)] select-none">
             {/* As-Of button */}
             <button
               type="button"
               onClick={handleAsOf}
               className={cn(
                 'px-2 py-0.5 rounded text-[10px] font-mono font-medium border transition-colors flex-shrink-0',
                 activeTopoMode === 'asof'
                   ? 'bg-[var(--accent)]/15 text-[var(--accent)] border-[var(--accent)]/30'
                   : 'text-[var(--text-faint)] border-transparent hover:text-[var(--text-secondary)]'
               )}
             >
               As-Of
             </button>

             {/* Timeline track */}
             <div className="relative flex-1 flex items-center">
               {/* Event pins */}
               {EVENT_PINS.map(({ day, label }) => (
                 <div
                   key={day}
                   className="absolute flex flex-col items-center pointer-events-none"
                   style={{ left: `${(day / WINDOW_DAYS) * 100}%`, transform: 'translateX(-50%)' }}
                   title={label}
                 >
                   <div className="w-1.5 h-1.5 rounded-full bg-[var(--status-warn)] opacity-70" />
                 </div>
               ))}
               {/* Range input */}
               <input
                 type="range"
                 min={0}
                 max={WINDOW_DAYS}
                 step={1}
                 defaultValue={currentDay}
                 onChange={handleChange}
                 className="w-full h-1.5 appearance-none rounded-full bg-[var(--surface-hover)] accent-[var(--accent)] cursor-pointer"
                 aria-label="Scrub topology timestamp"
               />
             </div>

             {/* Current day label */}
             <span className="text-[10px] font-mono text-[var(--text-faint)] flex-shrink-0 min-w-[64px] text-right">
               Day {currentDay} · {formatDayDate(currentDay)}
             </span>

             {/* Jump to Now */}
             <button
               type="button"
               onClick={handleJumpNow}
               className="px-2 py-0.5 rounded text-[10px] font-mono font-medium border border-[var(--border-subtle)] text-[var(--text-secondary)] hover:text-[var(--text-primary)] hover:border-[var(--border-strong)] transition-colors flex-shrink-0"
             >
               Now ⚡
             </button>
           </div>
         );
       };
       ```

       **Note on `defaultValue` vs `value`:** Using `defaultValue` (uncontrolled) for the range so the user's drag feels immediate; the debounced handler updates context state. This avoids visual lag from React re-rendering on every drag pixel. The `currentDay` derived value is only used for initial mount.

       TypeScript: `ReturnType<typeof setTimeout>` for the debounce ref type.
